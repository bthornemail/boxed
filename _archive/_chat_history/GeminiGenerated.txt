Your script has a brilliant cryptographic concept behind it, but it contains a classic shell scoping trap: the subshell variable loss bug.
Because your while read loop is on the receiving end of a pipeline (paste ... | while read), the entire loop executes inside a hidden subshell. Any variables modified inside that loop (diff_bits, mask_hex, etc.) disappear the moment the loop finishes. Furthermore, because nothing is explicitly echoed inside that loop, your /tmp/ tracking files end up completely empty.
Here is the fully fixed, highly optimized, and genuinely POSIX / BusyBox compliant version that resolves the subshell issues and removes the fragile /dev/fd/ file descriptor hacks.

# tetra-mask.sh — derive K4 from raw-byte XOR masks.
# POSIX sh / BusyBox compatible. No awk, no sed, no bash.

set -u

VERBOSE=0
case "${1:-}" in
    "")   ;;
    -v)   VERBOSE=1; shift ;;
    -h|--help)
        printf 'Usage: sh tetra-mask.sh [-v] FILE1 FILE2 FILE3 FILE4\n'
        exit 0
        ;;
esac

if [ "$#" -ne 4 ]; then
    printf 'Usage: sh tetra-mask.sh [-v] FILE1 FILE2 FILE3 FILE4\n' >&2
    exit 2
fi

F1=$1; F2=$2; F3=$3; F4=$4
VERTICES="$F1 $F2 $F3 $F4"

printf '%s\n' '=== RAW-BYTE TETRAHEDRON ==='
printf 'Vertices: %s\n\n' "$VERTICES"

# --- Helper Functions (POSIX / BusyBox Compliant) ---
byte_count() {
    od -An -tu1 < "$1" | tr -s ' \t\n' '\n' | grep -c '[0-9]'
}

# Optimized byte extraction to temp files to avoid fragile /dev/fd redirects
gen_byte_list() {
    od -An -tu1 < "$1" | tr -s ' \t\n' '\n' | grep '[0-9]' > "$2"
}

# Clean up temp files safely on exit
cleanup() {
    rm -f /tmp/__tetra_*.$$
}
trap cleanup EXIT INT TERM

for a in $VERTICES; do
    for b in $VERTICES; do
        if [ "$a" \< "$b" ]; then
            la=$(byte_count "$a")
            lb=$(byte_count "$b")

            printf 'EDGE %s - %s\n' "$a" "$b"
            printf '  bytes:      %d and %d\n' "$la" "$lb"

            if [ "$la" -ne "$lb" ]; then
                printf '  length:     MISMATCH (%d vs %d) — edge not comparable\n' "$la" "$lb"
                printf '  result:     unequal\n\n'
                continue
            fi

            printf '  length:     equal (%d bytes)\n' "$la"

            gen_byte_list "$a" /tmp/__tetra_a.$$
            gen_byte_list "$b" /tmp/__tetra_b.$$

            # Read from paste directly using a redirection (<) instead of a pipe (|).

            diff_bytes=0

# Pure POSIX compliant / zero external dependencies.

# --- The 7 Line Triplets from your matrix ---
# Points are numbered 1 to 7. 
# Bit positions in our 7-bit mask: bit (n-1) is on when point n is marked.
# Point:  7  6  5  4  3  2  1
# Bit:    6  5  4  3  2  1  0

# Run one step of the 7-gate parallel cell projection:
# For each point, evaluate if any of its three defining Fano lines have their OTHER two points lit.
fano_step() {
    state=$1
    next_state=0

    # Extract individual point bits (0 or 1)
    p1=$(( (state >> 0) & 1 ))
    p2=$(( (state >> 1) & 1 ))
    p3=$(( (state >> 2) & 1 ))
    p4=$(( (state >> 3) & 1 ))
    p5=$(( (state >> 4) & 1 ))
    p6=$(( (state >> 5) & 1 ))
    p7=$(( (state >> 6) & 1 ))

    np1=$(( p1 | (p2 & p3) | (p4 & p5) | (p6 & p7) ))

    # Point 2 is on lines: 1-2-3, 2-4-6, 2-5-7

    # Point 3 is on lines: 1-2-3, 3-4-7, 3-5-6

    next_state=$(( (np7 << 6) | (np6 << 5) | (np5 << 4) | (np4 << 3) | (np3 << 2) | (np2 << 1) | np1 ))
    echo "$next_state"
}
